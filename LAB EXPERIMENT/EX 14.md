#include <stdio.h>
#include <string.h>

char input[20], stack[20];
int ip = 0, top = -1;

void check()
{
    if (top >= 0 && (stack[top] == 'a' || stack[top] == 'b'))
    {
        stack[top] = 'E';
        stack[top + 1] = '\0';
        printf("\nReduce: E->a/b");
    }

    if (top >= 2 &&
        stack[top - 2] == 'E' &&
        (stack[top - 1] == '+' ||
         stack[top - 1] == '*' ||
         stack[top - 1] == '/') &&
        stack[top] == 'E')
    {
        char op = stack[top - 1];
        top -= 2;
        stack[top] = 'E';
        stack[top + 1] = '\0';

        if (op == '+')
            printf("\nReduce: E->E+E");
        else if (op == '*')
            printf("\nReduce: E->E*E");
        else
            printf("\nReduce: E->E/E");
    }
}

int main()
{
    int len;

    printf("SHIFT REDUCE PARSER\n");
    printf("Grammar:\nE->E+E\nE->E/E\nE->E*E\nE->a/b\n");

    printf("Enter the input string: ");
    scanf("%19s", input);

    len = strlen(input);

    printf("\nStack\tInput\tAction\n");

    while (ip < len)
    {
        stack[++top] = input[ip];
        stack[top + 1] = '\0';

        printf("$%s\t%s$\tShift %c\n",
               stack, input + ip + 1, input[ip]);

        ip++;
        check();
    }

    check();

    if (top == 0 && stack[0] == 'E')
        printf("\nString Accepted\n");
    else
        printf("\nString Rejected\n");

    return 0;
}
